# Dashboard-only dependencies for Streamlit Community Cloud.
# The dashboard talks ONLY to the deployed Render API over HTTP — it never imports
# the training stack (torch/prophet/xgboost/scikit-learn) or the DB layer.
#
# NOTE on Python version: pin the app to Python 3.11 in Streamlit Cloud
# ("Advanced settings" -> Python version). Community Cloud otherwise defaults to a
# very new Python (3.14) for which pandas/pillow have NO wheels yet, so pip tries
# to compile them from source and fails. These deps have wheels on Python 3.11-3.13.
#
# - streamlit is intentionally NOT pinned here: Community Cloud preinstalls a
#   Python-matched streamlit, and pinning streamlit==1.38.0 dragged in pillow<11
#   (whose wheels stop at cp313), which was the build failure on newer Python.
# - pandas allowed a patch range so a 3.13 build can still find a wheel (2.2.3);
#   the dashboard only tabulates API JSON, so a patch bump changes no displayed value.
pandas>=2.2.2,<2.3
plotly==5.23.0
requests==2.32.3
python-dotenv==1.0.1
